CREATE TABLE conta (
    id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    tipo VARCHAR(30) NOT NULL,
    saldo_inicial NUMERIC(12,2) NOT NULL DEFAULT 0,
    ativa BOOLEAN NOT NULL DEFAULT TRUE,

    CONSTRAINT ck_conta_saldo_inicial
        CHECK (saldo_inicial >= 0)
);

CREATE TABLE categoria (
    id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    tipo VARCHAR(20) NOT NULL,

    CONSTRAINT ck_categoria_tipo
        CHECK (tipo IN ('RECEITA', 'DESPESA'))
);

CREATE TABLE transacao (
    id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    descricao VARCHAR(255) NOT NULL,
    valor NUMERIC(12,2) NOT NULL,
    tipo VARCHAR(20) NOT NULL,
    data_transacao DATE NOT NULL,

    conta_id BIGINT NOT NULL,
    categoria_id BIGINT NOT NULL,

    CONSTRAINT ck_transacao_valor
        CHECK (valor > 0),

    CONSTRAINT ck_transacao_tipo
        CHECK (tipo IN ('RECEITA', 'DESPESA')),

    CONSTRAINT fk_transacao_conta
        FOREIGN KEY (conta_id)
        REFERENCES conta(id),

    CONSTRAINT fk_transacao_categoria
        FOREIGN KEY (categoria_id)
        REFERENCES categoria(id)
);

CREATE TABLE orcamento (
    id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    valor_limite NUMERIC(12,2) NOT NULL,
    mes SMALLINT NOT NULL,
    ano SMALLINT NOT NULL,

    categoria_id BIGINT NOT NULL,

    CONSTRAINT ck_orcamento_valor
        CHECK (valor_limite >= 0),

    CONSTRAINT ck_orcamento_mes
        CHECK (mes BETWEEN 1 AND 12),

    CONSTRAINT ck_orcamento_ano
        CHECK (ano >= 2000),

    CONSTRAINT fk_orcamento_categoria
        FOREIGN KEY (categoria_id)
        REFERENCES categoria(id),

    CONSTRAINT uk_orcamento_categoria_periodo
        UNIQUE (categoria_id, mes, ano)
);

CREATE TABLE teste (
    id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nome VARCHAR(100),
    descricao VARCHAR(255)
);
